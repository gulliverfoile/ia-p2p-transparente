# LECCION: El Laboratorio JS

Este plugin genera una bola que rebota. Si el motor es fluido, la animación no debería dar tirones al cambiar de lección.

PLUGIN:
const app = document.getElementById('app');
app.innerHTML = '<canvas id="c" width="400" height="200" style="background:#000; display:block; margin:auto;"></canvas>';
const ctx = document.getElementById('c').getContext('2d');
let x = 200, y = 100, dx = 2, dy = 2;
function draw() {
    ctx.clearRect(0,0,400,200);
    ctx.fillStyle = '#3b82f6';
    ctx.beginPath(); ctx.arc(x,y,10,0,Math.PI*2); ctx.fill();
    if(x+dx > 390 || x+dx < 10) dx = -dx;
    if(y+dy > 190 || y+dy < 10) dy = -dy;
    x+=dx; y+=dy;
    requestAnimationFrame(draw);
}
draw();
